// Run via verify/repro/c2_unclosed_listener.sh (copied to internal/xds/server/zz_c2_listener_probe_test.go on branch evalon/grpc-go-xd-bd84b6f8).
package server

import (
	"fmt"
	"net"
	"os"
	"runtime"
	"strconv"
	"syscall"
	"testing"
	"time"
)

// listeningTCPSockets returns the local addresses of every fd in this process
// that is a socket in the LISTEN state (SO_ACCEPTCONN == 1).
func listeningTCPSockets() []string {
	var out []string
	ents, _ := os.ReadDir("/proc/self/fd")
	for _, e := range ents {
		fd, err := strconv.Atoi(e.Name())
		if err != nil {
			continue
		}
		v, err := syscall.GetsockoptInt(fd, syscall.SOL_SOCKET, syscall.SO_ACCEPTCONN)
		if err != nil || v != 1 {
			continue
		}
		sa, err := syscall.Getsockname(fd)
		if err != nil {
			continue
		}
		if in4, ok := sa.(*syscall.SockaddrInet4); ok {
			out = append(out, fmt.Sprintf("fd=%d %s", fd, net.JoinHostPort(net.IP(in4.Addr[:]).String(), strconv.Itoa(in4.Port))))
		}
	}
	return out
}

// Runs after TestHandleRDSUpdate_ReleasesSupersededRouteConfiguration (method
// order is alphabetical) and reports sockets it left in the LISTEN state.
func (s) TestZZZ_C2ListenerProbe(t *testing.T) {
	socks := listeningTCPSockets()
	t.Logf("C2-PROBE listening sockets still open after previous test: %d %v", len(socks), socks)
	for _, sk := range socks {
		var fd int
		var addr string
		fmt.Sscanf(sk, "fd=%d %s", &fd, &addr)
		c, err := net.DialTimeout("tcp", addr, time.Second)
		t.Logf("C2-PROBE dial %s -> err=%v (nil means the port is still being listened on)", addr, err)
		if c != nil {
			c.Close()
		}
	}
	// Nuance: the net package attaches a GC finalizer to the socket. Show
	// whether a forced GC happens to reclaim it.
	for i := 0; i < 3; i++ {
		runtime.GC()
		time.Sleep(50 * time.Millisecond)
	}
	t.Logf("C2-PROBE listening sockets after 3 forced GC cycles: %d %v", len(listeningTCPSockets()), listeningTCPSockets())
}
