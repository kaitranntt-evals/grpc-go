package test

import (
	"context"
	"net"
	"testing"
	"time"

	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"
	"google.golang.org/grpc/encoding/gzip"
)

var _ = gzip.Name

func TestEvidenceC4DupNamePrecedence(t *testing.T) {
	events := make(chan string, 10)
	sd := &grpc.ServiceDesc{
		ServiceName: "dup.Svc",
		HandlerType: (*any)(nil),
		Methods: []grpc.MethodDesc{{
			MethodName: "Call",
			Handler: func(srv any, ctx context.Context, dec func(any) error, _ grpc.UnaryServerInterceptor) (any, error) {
				events <- "unaryHandler"
				var in []byte
				_ = dec(&in)
				return &in, nil
			},
		}},
		Streams: []grpc.StreamDesc{{
			StreamName: "Call",
			Handler: func(srv any, stream grpc.ServerStream) error {
				events <- "streamHandler"
				var in []byte
				_ = stream.RecvMsg(&in)
				return stream.SendMsg(&in)
			},
		}},
	}
	srv := grpc.NewServer(
		grpc.UnaryInterceptor(func(ctx context.Context, req any, info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (any, error) {
			events <- "unaryInterceptor"
			return handler(ctx, req)
		}),
		grpc.StreamInterceptor(func(srv any, ss grpc.ServerStream, info *grpc.StreamServerInfo, handler grpc.StreamHandler) error {
			events <- "streamInterceptor"
			return handler(srv, ss)
		}),
		grpc.ForceServerCodec(rawCodec{}),
	)
	srv.RegisterService(sd, struct{}{})
	lis, err := net.Listen("tcp", "localhost:0")
	if err != nil {
		t.Fatal(err)
	}
	go srv.Serve(lis)
	defer srv.Stop()

	cc, err := grpc.NewClient(lis.Addr().String(), grpc.WithTransportCredentials(insecure.NewCredentials()), grpc.WithDefaultCallOptions(grpc.ForceCodec(rawCodec{})))
	if err != nil {
		t.Fatal(err)
	}
	defer cc.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	in := []byte("hi")
	var out []byte
	err = cc.Invoke(ctx, "/dup.Svc/Call", &in, &out)
	t.Logf("EVIDENCE_C4 invoke err=%v", err)
	close(events)
	for e := range events {
		t.Logf("EVIDENCE_C4 event=%s", e)
	}
}

type rawCodec struct{}

func (rawCodec) Marshal(v any) ([]byte, error) { b := v.(*[]byte); return *b, nil }
func (rawCodec) Unmarshal(data []byte, v any) error {
	b := v.(*[]byte)
	*b = append([]byte(nil), data...)
	return nil
}
func (rawCodec) Name() string { return "evidence-raw" }
