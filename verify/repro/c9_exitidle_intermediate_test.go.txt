package endpointsharding_test

import (
	"fmt"
	"sync"
	"sync/atomic"
	"testing"

	"google.golang.org/grpc/balancer"
	"google.golang.org/grpc/balancer/endpointsharding"
	"google.golang.org/grpc/connectivity"
	"google.golang.org/grpc/resolver"
)

type c9CC struct {
	balancer.ClientConn
	mu        sync.Mutex
	completed *atomic.Int32
	log       []string
}

func (c *c9CC) UpdateState(s balancer.State) {
	var cs []string
	for _, ch := range endpointsharding.ChildStatesFromPicker(s.Picker) {
		cs = append(cs, fmt.Sprintf("%s=%v", ch.Endpoint.Addresses[0].Addr, ch.State.ConnectivityState))
	}
	c.mu.Lock()
	c.log = append(c.log, fmt.Sprintf("agg=%v children=%v childExitIdleCompleted=%d", s.ConnectivityState, cs, c.completed.Load()))
	c.mu.Unlock()
}

type c9Child struct {
	balancer.Balancer
	cc        balancer.ClientConn
	completed *atomic.Int32
}

func (c *c9Child) UpdateClientConnState(balancer.ClientConnState) error {
	c.cc.UpdateState(balancer.State{ConnectivityState: connectivity.Idle})
	return nil
}
func (c *c9Child) ResolverError(error) {}
func (c *c9Child) Close()              {}
func (c *c9Child) ExitIdle() {
	c.cc.UpdateState(balancer.State{ConnectivityState: connectivity.Connecting})
	c.completed.Add(1)
}

func TestVerifyC9ParentExitIdlePublications(t *testing.T) {
	completed := &atomic.Int32{}
	cc := &c9CC{completed: completed}
	builder := func(ccc balancer.ClientConn, _ balancer.BuildOptions) balancer.Balancer {
		return &c9Child{cc: ccc, completed: completed}
	}
	b := endpointsharding.NewBalancer(cc, balancer.BuildOptions{}, builder, endpointsharding.Options{DisableAutoReconnect: true})
	defer b.Close()
	var eps []resolver.Endpoint
	for _, a := range []string{"a", "b", "c"} {
		eps = append(eps, resolver.Endpoint{Addresses: []resolver.Address{{Addr: a}}})
	}
	if err := b.UpdateClientConnState(balancer.ClientConnState{ResolverState: resolver.State{Endpoints: eps}}); err != nil {
		t.Fatal(err)
	}
	cc.mu.Lock()
	before := len(cc.log)
	cc.mu.Unlock()
	b.ExitIdle()
	cc.mu.Lock()
	defer cc.mu.Unlock()
	got := cc.log[before:]
	for i, l := range got {
		t.Logf("parent notification %d during/after ExitIdle: %s", i+1, l)
	}
	if len(got) != 1 {
		t.Errorf("parent ExitIdle produced %d parent notifications, want 1 consolidated notification", len(got))
	}
}
